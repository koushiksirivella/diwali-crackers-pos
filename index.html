"YES" : "NO"]); }); sheet.getRange(1, 1, rows.length, headers.length).setValues(rows); return ContentService.createTextOutput(JSON.stringify({status: "ok", count: rows.length - 1})).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var values = sheet.getDataRange().getValues(); return ContentService.createTextOutput(JSON.stringify({rows: values})).setMimeType(ContentService.MimeType.JSON); }
